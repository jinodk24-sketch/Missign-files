# Task 4, Version 07: Unknown-Support Adaptive Posterior Audit
# =====================================================================
#
# PURPOSE
# -------
# Version 06 verified numerical convergence only inside a predeclared
# parameter rectangle. That is not sufficient for a deployed method because
# the learner does not know the true minimum and maximum worker capacity or
# resource aversion.
#
# Version 07 removes that assumption.
#
# The worker parameters remain positive and conceptually unbounded:
#
#     C > 0,
#     lambda > 0.
#
# We transform them to unconstrained log coordinates:
#
#     x = log(C / I_ref),
#     z = log(lambda / b).
#
# A broad Student-t prior is placed on (x, z). The numerical grid is initialized
# from a central prior-probability region, not from the true parameter range.
# During inference, the grid expands when:
#
#   1. exact effort observations leave no compatible numerical support; or
#   2. posterior mass accumulates near one or more grid boundaries.
#
# The grid spacing is preserved as the region expands, so expansion increases
# the number of numerical integration points instead of silently making the
# approximation coarser.
#
# SCIENTIFIC QUESTION
# -------------------
# Can effort-aware Bayesian prediction remain accurate when the learner is not
# given true parameter bounds?
#
# COMPARISONS
# -----------
# For each prior scale, true worker, and exact binary outcome sequence:
#
#   fixed_33:
#       33 x 33 log-parameter grid initialized from the prior and never expanded.
#
#   adaptive_33:
#       same initial 33 x 33 grid, but expands based only on posterior diagnostics.
#
#   reference_65:
#       same adaptive rule with a finer 65 x 65 initial resolution, a stricter
#       boundary threshold, and a larger computational budget. This is a
#       numerical reference, not an analytic ground truth.
#
# Each approximation is evaluated with:
#
#   outcome_only:
#       workloads and binary outcomes.
#
#   exact_spread2:
#       the same outcomes plus exact effort observations in rounds 1 and 3,
#       corresponding to workloads 6 and 8.
#
# PRIOR-SCALE SENSITIVITY
# -----------------------
# The script tests moderately broad, broad, and very broad Student-t priors.
# All priors are unbounded. Their numerical initial regions are constructed
# from the same central marginal prior mass.
#
# TRUE WORKERS
# ------------
# The deterministic evaluation workers include:
#
#   - workers comfortably inside the moderate initial region;
#   - workers near that region's numerical boundary;
#   - workers outside the moderate initial region;
#   - extreme workers intended to test expansion and prior sensitivity.
#
# These labels are used only for evaluation. The learner does not receive them.
#
# IMPORTANT
# ---------
# Capacity and resource aversion are continuous. Effort remains discrete:
#
#     E = {0, 0.05, ..., 1.00}.
#
# The script exactly enumerates all 2^5 binary outcome sequences. No Monte Carlo
# seed is used.
#
# OUTPUT
# ------
# The script creates a ZIP containing all CSV tables, configuration, summary,
# and figures. It prints percent progress, elapsed time, and ETA.
#
# Paste this entire file into one Google Colab cell and run it.

def stable_sigmoid(x):
    """Numerically stable logistic sigmoid."""
    x = np.asarray(x, dtype=float)
    output = np.empty_like(x)

    nonnegative = x >= 0.0
    output[nonnegative] = 1.0 / (
        1.0 + np.exp(-x[nonnegative])
    )

    exp_x = np.exp(x[~nonnegative])
    output[~nonnegative] = exp_x / (1.0 + exp_x)

    return output


def success_probability(effort, load_ratio):
    """
    Locked success model.

    p(0, rho) = 0
    p(e, rho) = sigmoid(beta * (e - rho)) for e > 0.
    """
    effort = np.asarray(effort, dtype=float)
    positive_probability = stable_sigmoid(
        BETA * (effort - load_ratio)
    )

    return np.where(
        effort <= 0.0,
        0.0,
        positive_probability,
    )


def choose_optimal_effort(workload, capacity, resource_aversion):
    """
    Deterministic best response with lowest-effort tie-breaking.
    """
    load_ratio = workload / capacity

    probabilities = success_probability(
        EFFORT_GRID,
        load_ratio,
    )

    utilities = (
        PAYMENT_PER_UNIT * probabilities
        - resource_aversion * EFFORT_GRID**2
    )

    maximum_utility = float(np.max(utilities))
    maximizing_indices = np.flatnonzero(
        np.isclose(
            utilities,
            maximum_utility,
            atol=ARGMAX_TOL,
            rtol=0.0,
        )
    )

    best_index = int(maximizing_indices[0])

    return (
        float(EFFORT_GRID[best_index]),
        float(probabilities[best_index]),
        maximum_utility,
    )


def compute_forward_matrices(capacity_vector, lambda_vector):
    """
    Vectorized worker behavior for every workload and numerical grid point.
    """
    capacity_vector = np.asarray(capacity_vector, dtype=float)
    lambda_vector = np.asarray(lambda_vector, dtype=float)

    candidate_count = len(capacity_vector)

    effort_matrix = np.zeros(
        (len(ALL_WORKLOADS), candidate_count),
        dtype=np.float32,
    )

    probability_matrix = np.zeros(
        (len(ALL_WORKLOADS), candidate_count),
        dtype=float,
    )

    effort_column = EFFORT_GRID[:, None]
    effort_squared_column = effort_column**2

    for workload_index, workload in enumerate(ALL_WORKLOADS):
        load_ratio_row = (
            workload / capacity_vector
        )[None, :]

        probabilities = stable_sigmoid(
            BETA * (effort_column - load_ratio_row)
        )

        probabilities[0, :] = 0.0

        utilities = (
            PAYMENT_PER_UNIT * probabilities
            - lambda_vector[None, :] * effort_squared_column
        )

        # np.argmax returns the first maximizer, implementing lowest-effort
        # tie-breaking because EFFORT_GRID is increasing.
        best_indices = np.argmax(utilities, axis=0)

        effort_matrix[workload_index, :] = EFFORT_GRID[
            best_indices
        ]

        probability_matrix[workload_index, :] = probabilities[
            best_indices,
            np.arange(candidate_count),
        ]

    return effort_matrix, probability_matrix

X_PRIOR_LOCATION = math.log(
    PRIOR_MEDIAN_CAPACITY / I_REFERENCE
)

Z_PRIOR_LOCATION = math.log(
    PRIOR_MEDIAN_LAMBDA / PAYMENT_PER_UNIT
)


def student_t_log_density(value, location, scale):
    """
    Log density of a location-scale Student-t distribution.
    """
    standardized = (value - location) / scale

    constant = (
        math.lgamma((PRIOR_DF + 1.0) / 2.0)
        - math.lgamma(PRIOR_DF / 2.0)
        - 0.5 * math.log(PRIOR_DF * math.pi)
        - math.log(scale)
    )

    return (
        constant
        - ((PRIOR_DF + 1.0) / 2.0)
        * np.log1p((standardized**2) / PRIOR_DF)
    )


def initial_log_bounds(prior_definition):
    """
    Initial numerical bounds from central marginal prior mass.

    No true parameter range is used.
    """
    lower_tail = (
        1.0 - INITIAL_MARGINAL_PRIOR_MASS
    ) / 2.0

    upper_tail = 1.0 - lower_tail

    lower_quantile = float(
        student_t_distribution.ppf(
            lower_tail,
            df=PRIOR_DF,
        )
    )

    upper_quantile = float(
        student_t_distribution.ppf(
            upper_tail,
            df=PRIOR_DF,
        )
    )

    x_lower = (
        X_PRIOR_LOCATION
        + prior_definition["x_scale"] * lower_quantile
    )

    x_upper = (
        X_PRIOR_LOCATION
        + prior_definition["x_scale"] * upper_quantile
    )

    z_lower = (
        Z_PRIOR_LOCATION
        + prior_definition["z_scale"] * lower_quantile
    )

    z_upper = (
        Z_PRIOR_LOCATION
        + prior_definition["z_scale"] * upper_quantile
    )

    return {
        "x_lower": float(x_lower),
        "x_upper": float(x_upper),
        "z_lower": float(z_lower),
        "z_upper": float(z_upper),
    }


def log_bounds_to_original(bounds):
    return {
        "capacity_lower": float(
            I_REFERENCE * math.exp(bounds["x_lower"])
        ),
        "capacity_upper": float(
            I_REFERENCE * math.exp(bounds["x_upper"])
        ),
        "lambda_lower": float(
            PAYMENT_PER_UNIT * math.exp(bounds["z_lower"])
        ),
        "lambda_upper": float(
            PAYMENT_PER_UNIT * math.exp(bounds["z_upper"])
        ),
    }

GRID_CACHE = OrderedDict()


def grid_cache_key(
    bounds,
    target_dx,
    target_dz,
    max_grid_points,
):
    return (
        round(bounds["x_lower"], 12),
        round(bounds["x_upper"], 12),
        round(bounds["z_lower"], 12),
        round(bounds["z_upper"], 12),
        round(target_dx, 12),
        round(target_dz, 12),
        int(max_grid_points),
    )


def build_grid_payload(
    bounds,
    target_dx,
    target_dz,
    max_grid_points,
):
    """
    Build or retrieve one midpoint quadrature grid.

    Grid spacing is never allowed to become coarser than the initial target.
    """
    key = grid_cache_key(
        bounds,
        target_dx,
        target_dz,
        max_grid_points,
    )

    if key in GRID_CACHE:
        payload = GRID_CACHE.pop(key)
        GRID_CACHE[key] = payload
        return payload

    x_span = bounds["x_upper"] - bounds["x_lower"]
    z_span = bounds["z_upper"] - bounds["z_lower"]

    n_x = max(
        3,
        int(math.ceil(x_span / target_dx)),
    )

    n_z = max(
        3,
        int(math.ceil(z_span / target_dz)),
    )

    candidate_count = n_x * n_z

    if candidate_count > max_grid_points:
        return None

    actual_dx = x_span / n_x
    actual_dz = z_span / n_z

    x_points = (
        bounds["x_lower"]
        + (np.arange(n_x, dtype=float) + 0.5) * actual_dx
    )

    z_points = (
        bounds["z_lower"]
        + (np.arange(n_z, dtype=float) + 0.5) * actual_dz
    )

    x_mesh, z_mesh = np.meshgrid(
        x_points,
        z_points,
        indexing="ij",
    )

    x_vector = x_mesh.ravel()
    z_vector = z_mesh.ravel()

    capacity_vector = (
        I_REFERENCE * np.exp(x_vector)
    )

    lambda_vector = (
        PAYMENT_PER_UNIT * np.exp(z_vector)
    )

    effort_matrix, probability_matrix = (
        compute_forward_matrices(
            capacity_vector,
            lambda_vector,
        )
    )

    payload = {
        "bounds": dict(bounds),
        "n_x": n_x,
        "n_z": n_z,
        "candidate_count": candidate_count,
        "actual_dx": actual_dx,
        "actual_dz": actual_dz,
        "x_vector": x_vector,
        "z_vector": z_vector,
        "capacity_vector": capacity_vector,
        "lambda_vector": lambda_vector,
        "effort_matrix": effort_matrix,
        "probability_matrix": probability_matrix,
    }

    GRID_CACHE[key] = payload

    while len(GRID_CACHE) > GRID_CACHE_MAX_ENTRIES:
        GRID_CACHE.popitem(last=False)

    return payload

def normalize_log_weights(log_weights):
    log_weights = np.asarray(log_weights, dtype=float)
    finite_mask = np.isfinite(log_weights)

    if not np.any(finite_mask):
        return None

    maximum = float(np.max(log_weights[finite_mask]))

    shifted = np.zeros_like(log_weights)
    shifted[finite_mask] = np.exp(
        log_weights[finite_mask] - maximum
    )

    total = float(np.sum(shifted))

    if total <= 0.0 or not np.isfinite(total):
        return None

    return shifted / total


def compute_log_posterior(
    payload,
    prior_definition,
    history_workloads,
    history_outcomes,
    observed_efforts_by_round,
):
    """
    Full-history posterior on the current numerical grid.
    """
    log_weights = (
        student_t_log_density(
            payload["x_vector"],
            X_PRIOR_LOCATION,
            prior_definition["x_scale"],
        )
        + student_t_log_density(
            payload["z_vector"],
            Z_PRIOR_LOCATION,
            prior_definition["z_scale"],
        )
    )

    support_mask = np.ones(
        payload["candidate_count"],
        dtype=bool,
    )

    for history_index, (
        round_number,
        workload,
        outcome,
    ) in enumerate(
        zip(
            range(1, len(history_workloads) + 1),
            history_workloads,
            history_outcomes,
        )
    ):
        workload_index = WORKLOAD_TO_INDEX[
            float(workload)
        ]

        candidate_probabilities = payload[
            "probability_matrix"
        ][workload_index]

        if int(outcome) == 1:
            with np.errstate(divide="ignore"):
                log_weights += np.log(
                    candidate_probabilities
                )
        else:
            with np.errstate(divide="ignore", invalid="ignore"):
                log_weights += np.log1p(
                    -candidate_probabilities
                )

        if round_number in observed_efforts_by_round:
            observed_effort = observed_efforts_by_round[
                round_number
            ]

            support_mask &= np.isclose(
                payload["effort_matrix"][
                    workload_index
                ],
                observed_effort,
                atol=1e-12,
                rtol=0.0,
            )

    log_weights = log_weights.copy()
    log_weights[~support_mask] = -np.inf

    posterior = normalize_log_weights(log_weights)

    return posterior, support_mask


def boundary_masses(
    posterior,
    n_x,
    n_z,
    boundary_layers,
):
    matrix = posterior.reshape(n_x, n_z)

    layers_x = min(boundary_layers, n_x)
    layers_z = min(boundary_layers, n_z)

    return {
        "x_lower": float(
            np.sum(matrix[:layers_x, :])
        ),
        "x_upper": float(
            np.sum(matrix[-layers_x:, :])
        ),
        "z_lower": float(
            np.sum(matrix[:, :layers_z])
        ),
        "z_upper": float(
            np.sum(matrix[:, -layers_z:])
        ),
    }


def expand_bounds(bounds, directions):
    """
    Expand indicated sides while keeping all existing points inside.
    """
    expanded = dict(bounds)

    x_span = bounds["x_upper"] - bounds["x_lower"]
    z_span = bounds["z_upper"] - bounds["z_lower"]

    if "x_lower" in directions:
        expanded["x_lower"] -= (
            EXPANSION_SIDE_FRACTION * x_span
        )

    if "x_upper" in directions:
        expanded["x_upper"] += (
            EXPANSION_SIDE_FRACTION * x_span
        )

    if "z_lower" in directions:
        expanded["z_lower"] -= (
            EXPANSION_SIDE_FRACTION * z_span
        )

    if "z_upper" in directions:
        expanded["z_upper"] += (
            EXPANSION_SIDE_FRACTION * z_span
        )

    return expanded


def bounds_are_numerically_safe(bounds):
    values = [
        bounds["x_lower"],
        bounds["x_upper"],
        bounds["z_lower"],
        bounds["z_upper"],
    ]

    return all(
        np.isfinite(value)
        and abs(value) <= HARD_LOG_ABS_BOUND
        for value in values
    )


def fit_history(
    history_workloads,
    history_outcomes,
    observed_efforts_by_round,
    prior_definition,
    approximation_definition,
    bounds,
    target_dx,
    target_dz,
    expansion_events_used,
):
    """
    Fit the current full history, expanding the grid when required.
    """
    adaptive = approximation_definition["adaptive"]
    threshold = approximation_definition[
        "boundary_mass_threshold"
    ]

    max_expansions = approximation_definition[
        "max_expansion_events"
    ]

    boundary_layers = approximation_definition[
        "boundary_layers"
    ]

    max_grid_points = approximation_definition[
        "max_grid_points"
    ]

    collapse_expansion_events = 0
    boundary_expansion_events = 0
    adaptation_limit_reached = False

    while True:
        if not bounds_are_numerically_safe(bounds):
            return {
                "valid": False,
                "failure_reason": "hard_log_bound",
                "bounds": bounds,
                "expansion_events_used": expansion_events_used,
                "collapse_expansion_events": (
                    collapse_expansion_events
                ),
                "boundary_expansion_events": (
                    boundary_expansion_events
                ),
            }

        payload = build_grid_payload(
            bounds,
            target_dx,
            target_dz,
            max_grid_points,
        )

        if payload is None:
            return {
                "valid": False,
                "failure_reason": "grid_point_budget",
                "bounds": bounds,
                "expansion_events_used": expansion_events_used,
                "collapse_expansion_events": (
                    collapse_expansion_events
                ),
                "boundary_expansion_events": (
                    boundary_expansion_events
                ),
            }

        posterior, support_mask = compute_log_posterior(
            payload,
            prior_definition,
            history_workloads,
            history_outcomes,
            observed_efforts_by_round,
        )

        support_size = int(np.sum(support_mask))

        if posterior is None:
            if (
                not adaptive
                or expansion_events_used >= max_expansions
            ):
                return {
                    "valid": False,
                    "failure_reason": "posterior_support_collapse",
                    "bounds": bounds,
                    "payload": payload,
                    "support_size": support_size,
                    "expansion_events_used": expansion_events_used,
                    "collapse_expansion_events": (
                        collapse_expansion_events
                    ),
                    "boundary_expansion_events": (
                        boundary_expansion_events
                    ),
                }

            bounds = expand_bounds(
                bounds,
                {
                    "x_lower",
                    "x_upper",
                    "z_lower",
                    "z_upper",
                },
            )

            expansion_events_used += 1
            collapse_expansion_events += 1
            continue

        masses = boundary_masses(
            posterior,
            payload["n_x"],
            payload["n_z"],
            boundary_layers,
        )

        if adaptive:
            directions = {
                side
                for side, mass in masses.items()
                if mass > threshold
            }

            if directions:
                if expansion_events_used >= max_expansions:
                    adaptation_limit_reached = True
                else:
                    bounds = expand_bounds(
                        bounds,
                        directions,
                    )

                    expansion_events_used += 1
                    boundary_expansion_events += 1
                    continue

        predicted_curve = (
            payload["probability_matrix"] @ posterior
        )

        effective_sample_size = float(
            1.0 / np.sum(posterior**2)
        )

        return {
            "valid": True,
            "failure_reason": "",
            "bounds": bounds,
            "payload": payload,
            "posterior": posterior,
            "predicted_curve": predicted_curve,
            "support_size": support_size,
            "boundary_masses": masses,
            "effective_sample_size": effective_sample_size,
            "expansion_events_used": expansion_events_used,
            "collapse_expansion_events": (
                collapse_expansion_events
            ),
            "boundary_expansion_events": (
                boundary_expansion_events
            ),
            "adaptation_limit_reached": (
                adaptation_limit_reached
            ),
        }


def run_sequential_inference(
    outcomes,
    true_training_efforts,
    method_name,
    prior_definition,
    approximation_definition,
):
    """
    Online inference over five rounds.

    Expansion decisions use only observations available up to the current round.
    """
    bounds = initial_log_bounds(prior_definition)

    initial_x_span = (
        bounds["x_upper"] - bounds["x_lower"]
    )

    initial_z_span = (
        bounds["z_upper"] - bounds["z_lower"]
    )

    initial_n = approximation_definition[
        "initial_points_per_dimension"
    ]

    target_dx = initial_x_span / initial_n
    target_dz = initial_z_span / initial_n

    history_workloads = []
    history_outcomes = []
    observed_efforts_by_round = {}

    expansion_events_used = 0
    total_collapse_expansions = 0
    total_boundary_expansions = 0

    final_state = None

    for round_index in range(HORIZON):
        round_number = round_index + 1

        history_workloads.append(
            float(
                TRAINING_WORKLOAD_SEQUENCE[
                    round_index
                ]
            )
        )

        history_outcomes.append(
            int(outcomes[round_index])
        )

        if (
            method_name == "exact_spread2"
            and round_number in MONITORED_ROUNDS
        ):
            observed_efforts_by_round[
                round_number
            ] = float(
                true_training_efforts[
                    round_index
                ]
            )

        state = fit_history(
            history_workloads=history_workloads,
            history_outcomes=history_outcomes,
            observed_efforts_by_round=observed_efforts_by_round,
            prior_definition=prior_definition,
            approximation_definition=approximation_definition,
            bounds=bounds,
            target_dx=target_dx,
            target_dz=target_dz,
            expansion_events_used=expansion_events_used,
        )

        total_collapse_expansions += state.get(
            "collapse_expansion_events",
            0,
        )

        total_boundary_expansions += state.get(
            "boundary_expansion_events",
            0,
        )

        expansion_events_used = state[
            "expansion_events_used"
        ]

        bounds = state["bounds"]

        if not state["valid"]:
            return {
                **state,
                "failed_round": round_number,
                "total_collapse_expansions": (
                    total_collapse_expansions
                ),
                "total_boundary_expansions": (
                    total_boundary_expansions
                ),
                "target_dx": target_dx,
                "target_dz": target_dz,
            }

        final_state = state

    original_bounds = log_bounds_to_original(
        final_state["bounds"]
    )

    maximum_boundary_mass = max(
        final_state["boundary_masses"].values()
    )

    return {
        **final_state,
        "failed_round": math.nan,
        "total_collapse_expansions": (
            total_collapse_expansions
        ),
        "total_boundary_expansions": (
            total_boundary_expansions
        ),
        "target_dx": target_dx,
        "target_dz": target_dz,
        "maximum_boundary_mass": (
            maximum_boundary_mass
        ),
        **original_bounds,
    }

def bernoulli_kl(
    true_probability,
    predicted_probability,
):
    p = np.clip(
        np.asarray(true_probability, dtype=float),
        PROBABILITY_CLIP,
        1.0 - PROBABILITY_CLIP,
    )

    q = np.clip(
        np.asarray(predicted_probability, dtype=float),
        PROBABILITY_CLIP,
        1.0 - PROBABILITY_CLIP,
    )

    return (
        p * np.log(p / q)
        + (1.0 - p) * np.log((1.0 - p) / (1.0 - q))
    )


def posterior_entropy(posterior):
    positive = posterior > 0.0

    return float(
        -np.sum(
            posterior[positive]
            * np.log(posterior[positive])
        )
    )


def compute_metrics(
    inference_result,
    true_curve,
    true_capacity,
    true_lambda,
):
    posterior = inference_result["posterior"]
    payload = inference_result["payload"]
    predicted_curve = inference_result[
        "predicted_curve"
    ]

    metrics = {
        "posterior_entropy": posterior_entropy(posterior),
        "normalized_posterior_entropy": (
            posterior_entropy(posterior)
            / math.log(len(posterior))
        ),
        "posterior_mean_capacity": float(
            np.dot(
                posterior,
                payload["capacity_vector"],
            )
        ),
        "posterior_mean_lambda": float(
            np.dot(
                posterior,
                payload["lambda_vector"],
            )
        ),
        "posterior_effective_sample_size": (
            inference_result[
                "effective_sample_size"
            ]
        ),
    }

    metrics["capacity_absolute_error"] = abs(
        metrics["posterior_mean_capacity"]
        - true_capacity
    )

    metrics["lambda_absolute_error"] = abs(
        metrics["posterior_mean_lambda"]
        - true_lambda
    )

    for evaluation_name, indices in EVALUATION_INDICES.items():
        target = true_curve[indices]
        predicted = predicted_curve[indices]

        absolute_error = np.abs(
            predicted - target
        )

        squared_error = (
            predicted - target
        ) ** 2

        kl_values = bernoulli_kl(
            target,
            predicted,
        )

        metrics[f"{evaluation_name}_mae"] = float(
            np.mean(absolute_error)
        )

        metrics[f"{evaluation_name}_rmse"] = float(
            np.sqrt(np.mean(squared_error))
        )

        metrics[
            f"{evaluation_name}_mean_kl"
        ] = float(np.mean(kl_values))

        metrics[
            f"{evaluation_name}_max_abs_error"
        ] = float(np.max(absolute_error))

    return metrics


def sequence_probability(
    outcomes,
    true_probabilities,
):
    probability = 1.0

    for outcome, success_probability_value in zip(
        outcomes,
        true_probabilities,
    ):
        probability *= (
            success_probability_value
            if outcome == 1
            else 1.0 - success_probability_value
        )

    return float(probability)

class ProgressTracker:
    def __init__(
        self,
        total_units,
        report_every_percent=1.0,
    ):
        self.total_units = float(total_units)
        self.completed_units = 0.0
        self.report_every_percent = float(
            report_every_percent
        )
        self.next_report_percent = 0.0
        self.start_time = time.perf_counter()

    @staticmethod
    def format_seconds(seconds):
        if not np.isfinite(seconds):
            return "unknown"

        seconds = max(0, int(round(seconds)))
        hours, remainder = divmod(seconds, 3600)
        minutes, seconds = divmod(remainder, 60)

        if hours > 0:
            return (
                f"{hours:d}h "
                f"{minutes:02d}m "
                f"{seconds:02d}s"
            )

        if minutes > 0:
            return f"{minutes:d}m {seconds:02d}s"

        return f"{seconds:d}s"

    def update(self, units=1.0, context=""):
        self.completed_units += float(units)

        percent = min(
            100.0,
            100.0
            * self.completed_units
            / self.total_units,
        )

        if (
            percent + 1e-12
            < self.next_report_percent
            and percent < 100.0
        ):
            return

        elapsed = time.perf_counter() - self.start_time

        if self.completed_units > 0.0:
            estimated_total = (
                elapsed
                * self.total_units
                / self.completed_units
            )

            eta = max(
                0.0,
                estimated_total - elapsed,
            )
        else:
            eta = math.inf

        suffix = f" | {context}" if context else ""

        print(
            f"Progress: {percent:6.2f}%"
            f" | elapsed {self.format_seconds(elapsed)}"
            f" | ETA {self.format_seconds(eta)}"
            f"{suffix}",
            flush=True,
        )

        while (
            self.next_report_percent
            <= percent + 1e-12
        ):
            self.next_report_percent += (
                self.report_every_percent
            )

                    "true_type_id": true_type_id,
                    "test_category": true_row[
                        "test_category"
                    ],
                    "true_capacity": float(
                        true_row["capacity"]
                    ),
                    "true_lambda": float(
                        true_row["lambda"]
                    ),
                    "outcome_sequence": "".join(
                        str(value)
                        for value in outcomes
                    ),
                    "sequence_probability": (
                        sequence_weight
                    ),
                    "approximation": (
                        approximation_name
                    ),
                    "baseline_valid": baseline[
                        "valid"
                    ],
                    "effort_aware_valid": (
                        effort_aware["valid"]
                    ),
                    "both_valid": (
                        baseline["valid"]
                        and effort_aware["valid"]
                    ),
                }

                if pair_row["both_valid"]:
                    for metric_name in [
                        "heldout_near_rmse",
                        "heldout_near_mean_kl",
                        "heldout_full_rmse",
                        "all_workloads_rmse",
                    ]:
                        improvement = (
                            baseline[metric_name]
                            - effort_aware[metric_name]
                        )

                        pair_row[
                            f"{metric_name}_improvement"
                        ] = improvement

                        pair_row[
                            f"{metric_name}_better"
                        ] = (
                            improvement
                            > METRIC_TIE_TOL
                        )

                        pair_row[
                            f"{metric_name}_worse"
                        ] = (
                            improvement
                            < -METRIC_TIE_TOL
                        )

                        pair_row[
                            f"{metric_name}_tie"
                        ] = (
                            abs(improvement)
                            <= METRIC_TIE_TOL
                        )

                paired_result_rows.append(pair_row)

        assert np.isclose(
            probability_mass_check,
            1.0,
            atol=1e-10,
        ), (
            f"Outcome probability did not sum to one for "
            f"{prior_name}, {true_type_id}: "
            f"{probability_mass_check}"
        )


progress_tracker.update(
    units=max(
        0.0,
        TOTAL_INFERENCE_RUNS
        - progress_tracker.completed_units,
    )
)

print(
    "Exact enumeration complete.",
    flush=True,
)

    "prior_name",
    "true_type_id",
    "test_category",
    "true_capacity",
    "true_lambda",
    "approximation",
    "adaptive",
    "is_reference",
    "method",
]

for group_key, group in sequence_results_df.groupby(
    group_columns,
    dropna=False,
):
    group_info = dict(
        zip(group_columns, group_key)
    )

    valid_group = group[group["valid"]].copy()

    valid_probability_mass = float(
        valid_group["sequence_probability"].sum()
    )

    collapse_probability = (
        1.0 - valid_probability_mass
    )

    row = {
        **group_info,
        "valid_probability_mass": (
            valid_probability_mass
        ),
        "collapse_probability": (
            collapse_probability
        ),
        "penalized_heldout_near_rmse": (
            collapse_probability
            * COLLAPSE_RMSE_PENALTY
        ),
    }

    # Expansion/runtime summaries include failed sequences as well.
    for column in [
        "runtime_seconds",
        "expansion_events_used",
        "collapse_expansion_events",
        "boundary_expansion_events",
    ]:
        row[f"expected_{column}"] = float(
            np.sum(
                group["sequence_probability"]
                * group[column].fillna(0.0)
            )
        )

    row[
        "adaptation_limit_probability"
    ] = float(
        np.sum(
            group["sequence_probability"]
            * group[
                "adaptation_limit_reached"
            ].astype(float)
        )
    )

    if valid_probability_mass > 0.0:
        for column in (
            EXPECTED_METRIC_COLUMNS
            + [
                "reference_all_curve_rmse",
                "reference_heldout_near_curve_rmse",
                "reference_heldout_full_curve_rmse",
                "final_grid_points",
                "final_n_x",
                "final_n_z",
                "final_support_size",
                "maximum_boundary_mass",
                "capacity_lower",
                "capacity_upper",
                "lambda_lower",
                "lambda_upper",
            ]
        ):
            weighted_value = float(
                np.sum(
                    valid_group[
                        "sequence_probability"
                    ]
                    * valid_group[column]
                )
                / valid_probability_mass
            )

            row[column] = weighted_value

        row[
            "penalized_heldout_near_rmse"
        ] += (
            valid_probability_mass
            * row["heldout_near_rmse"]
        )

    per_type_rows.append(row)
