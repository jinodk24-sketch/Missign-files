# Task 4, Version 06: Continuous-Type Posterior Convergence Audit
# =================================================================
#
# PURPOSE
# -------
# The worker's latent capacity C and resource aversion lambda are conceptually
# continuous:
#
#     C in [8, 16]
#     lambda in [0.4, 1.2]
#
# The worker's chosen effort remains discrete:
#
#     E = {0, 0.05, 0.10, ..., 1.00}.
#
# Earlier experiments used only 25 candidate parameter pairs. Version 06 tests
# whether effort-aware predictive learning persists when the posterior over
# continuous (C, lambda) is approximated by increasingly fine numerical
# quadrature grids.
#
# SCIENTIFIC QUESTIONS
# --------------------
#   1. Does exact effort-aware inference still improve held-out reliability
#      prediction for true workers with ordinary continuous parameter values?
#
#   2. Does support collapse disappear as the posterior approximation becomes
#      sufficiently fine?
#
#   3. Do posterior-predictive curves converge toward a high-resolution
#      reference approximation?
#
#   4. Are conclusions stable when the quadrature points are shifted inside
#      their cells?
#
#   5. What grid size is accurate enough for later multi-worker simulation?
#
# IMPORTANT MODEL DISTINCTION
# ---------------------------
#   - C and lambda are continuous latent worker characteristics.
#   - e is a discrete worker decision.
#   - The numerical grid does NOT define a finite catalogue of worker types.
#     It approximates integrals over a continuous parameter rectangle.
#
# BASELINE ASSUMPTIONS IN THIS AUDIT
# ----------------------------------
#   - deterministic rational best response;
#   - exact effort observation;
#   - no behavioral lapse;
#   - no measurement noise;
#   - known beta and payment rate;
#   - uniform prior over the continuous parameter rectangle;
#   - fixed workload sequence [6, 7, 8, 7, 6];
#   - exact enumeration of all 2^5 binary outcome sequences.
#
# METHODS
# -------
#   outcome_only:
#       uses workloads and binary outcomes.
#
#   exact_spread2:
#       additionally observes exact effort in rounds 1 and 3,
#       corresponding to workloads {6, 8}.
#
#   exact_unique3:
#       additionally observes exact effort in rounds 1, 2, and 3,
#       corresponding to workloads {6, 7, 8}.
#
# NUMERICAL APPROXIMATIONS
# ------------------------
# Midpoint grids:
#
#       5 x 5   =   25 points
#       9 x 9   =   81 points
#      17 x 17  =  289 points
#      33 x 33  = 1089 points
#      65 x 65  = 4225 points  (reference)
#
# Shifted 17 x 17 and 33 x 33 grids are also tested. The shifted points remain
# inside the same equal-area cells, so each point retains equal prior weight.
#
# TRUE WORKERS
# ------------
# The script uses 25 deterministic continuous true parameter pairs. These values
# do not coincide exactly with any tested quadrature point. There is no random
# seed and no Monte Carlo sampling.
#
# PRIMARY OUTPUTS
# ---------------
#   - support-collapse rate;
#   - held-out predictive RMSE, MAE, and Bernoulli KL;
#   - predictive-curve RMSE relative to the 4225-point reference posterior;
#   - paired improvement over outcome-only on the same outcome sequence;
#   - phase-shift sensitivity;
#   - runtime and approximate array memory;
#   - an automatic convergence-screening table.
#
# INTERPRETING COLLAPSE
# ---------------------
# Literal exact matching can leave a coarse point grid with no compatible
# quadrature point even though a compatible continuous parameter region exists.
# Such cases are recorded as numerical support collapse.
#
# Predictive metrics for an exact method are reported conditionally over true
# workers for which its numerical support does not collapse. A separate
# conservative metric assigns RMSE=1 to collapsed cases.
#
# OUTPUT PACKAGE
# --------------
# The script creates a ZIP containing all CSV tables, configuration, summary,
# and figures. Google Colab should download it automatically.
#
# Paste this entire file into one Google Colab cell and run it.

def stable_sigmoid(x):
    """Numerically stable logistic sigmoid."""
    x = np.asarray(x, dtype=float)
    output = np.empty_like(x)

    nonnegative = x >= 0.0
    output[nonnegative] = 1.0 / (
        1.0 + np.exp(-x[nonnegative])
    )

    exp_x = np.exp(x[~nonnegative])
    output[~nonnegative] = exp_x / (1.0 + exp_x)

    return output


def success_probability(effort, load_ratio):
    """
    Locked success model.

    p(0, rho) = 0
    p(e, rho) = sigmoid(beta * (e - rho)) for e > 0
    """
    effort = np.asarray(effort, dtype=float)
    positive_probability = stable_sigmoid(
        BETA * (effort - load_ratio)
    )

    return np.where(
        effort <= 0.0,
        0.0,
        positive_probability,
    )


def per_unit_worker_utility(
    effort,
    workload,
    capacity,
    resource_aversion,
):
    """
    Per-unit expected worker utility:
        U(e) = b p(e, I/C) - lambda e^2.
    """
    effort = np.asarray(effort, dtype=float)
    load_ratio = workload / capacity

    return (
        PAYMENT_PER_UNIT
        * success_probability(effort, load_ratio)
        - resource_aversion * effort**2
    )


def choose_optimal_effort(
    workload,
    capacity,
    resource_aversion,
):
    """
    Deterministic best response with lowest-effort tie-breaking.
    """
    utilities = per_unit_worker_utility(
        EFFORT_GRID,
        workload,
        capacity,
        resource_aversion,
    )

    maximum_utility = float(np.max(utilities))
    maximizing_indices = np.flatnonzero(
        np.isclose(
            utilities,
            maximum_utility,
            atol=ARGMAX_TOL,
            rtol=0.0,
        )
    )

    best_index = int(maximizing_indices[0])
    best_effort = float(EFFORT_GRID[best_index])

    best_probability = float(
        success_probability(
            np.array([best_effort]),
            workload / capacity,
        )[0]
    )

    return best_effort, best_probability, maximum_utility


def true_type_id(capacity, resource_aversion):
    """Readable deterministic identifier."""
    return (
        f"true_C{capacity:05.2f}_L{resource_aversion:04.2f}"
        .replace(".", "p")
    )

# 3. QUADRATURE GRID CONSTRUCTION
# =================================================================

def make_cell_phase_points(
    lower,
    upper,
    cell_count,
    phase,
):
    """
    One quadrature point inside each equal-width cell.

    phase=0.5 is the midpoint rule.
    phase=0.25 and 0.75 produce shifted within-cell rules.
    """
    cell_width = (upper - lower) / cell_count

    points = lower + (
        np.arange(cell_count, dtype=float) + phase
    ) * cell_width

    return points, cell_width


def build_approximation(definition):
    """
    Construct one equal-weight quadrature approximation.
    """
    capacity_points, capacity_cell_width = (
        make_cell_phase_points(
            CAPACITY_MIN,
            CAPACITY_MAX,
            definition["n_capacity"],
            definition["phase"],
        )
    )

    lambda_points, lambda_cell_width = (
        make_cell_phase_points(
            LAMBDA_MIN,
            LAMBDA_MAX,
            definition["n_lambda"],
            definition["phase"],
        )
    )

    capacity_mesh, lambda_mesh = np.meshgrid(
        capacity_points,
        lambda_points,
        indexing="ij",
    )

    capacity_vector = capacity_mesh.ravel()
    lambda_vector = lambda_mesh.ravel()
    candidate_count = len(capacity_vector)

    effort_matrix = np.zeros(
        (len(ALL_WORKLOADS), candidate_count),
        dtype=float,
    )

    probability_matrix = np.zeros(
        (len(ALL_WORKLOADS), candidate_count),
        dtype=float,
    )

    utility_matrix = np.zeros(
        (len(ALL_WORKLOADS), candidate_count),
        dtype=float,
    )

    # Vectorization is over candidate types; the effort grid is only 21 values.
    for workload_index, workload in enumerate(ALL_WORKLOADS):
        for candidate_index in range(candidate_count):
            effort, probability, utility = choose_optimal_effort(
                workload,
                capacity_vector[candidate_index],
                lambda_vector[candidate_index],
            )

            effort_matrix[
                workload_index,
                candidate_index,
            ] = effort

            probability_matrix[
                workload_index,
                candidate_index,
            ] = probability

            utility_matrix[
                workload_index,
                candidate_index,
            ] = utility

    prior = np.full(
        candidate_count,
        1.0 / candidate_count,
        dtype=float,
    )

    return {
        "definition": definition,
        "capacity_points": capacity_points,
        "lambda_points": lambda_points,
        "capacity_cell_width": capacity_cell_width,
        "lambda_cell_width": lambda_cell_width,
        "capacity_vector": capacity_vector,
        "lambda_vector": lambda_vector,
        "candidate_count": candidate_count,
        "effort_matrix": effort_matrix,
        "probability_matrix": probability_matrix,
        "utility_matrix": utility_matrix,
        "prior": prior,
        "log_prior": np.log(prior),
    }

    assert HORIZON == 5

    for monitored_rounds in MONITOR_SCHEDULES.values():
        assert all(

        )

    assert set(HELDOUT_NEAR_WORKLOADS.tolist()).isdisjoint(
        set(TRAINING_WORKLOAD_SEQUENCE.tolist())
    )

    approximation_names = [
        item["approximation"]
        for item in APPROXIMATION_DEFINITIONS
    ]

    assert len(approximation_names) == len(
        set(approximation_names)
    )

    assert REFERENCE_APPROXIMATION in approximation_names

    reference_count = sum(
        item["is_reference"]
        for item in APPROXIMATION_DEFINITIONS

    assert reference_count == 1

    p_zero = success_probability(np.array([0.0]), 0.75)[0]
    assert np.isclose(p_zero, 0.0)

    p_at_capacity = success_probability(
        np.array([1.0]),

    calibrated_effort, _, _ = choose_optimal_effort(
        workload=9.0,
        capacity=12.0,
        resource_aversion=0.8,
    )

    assert np.isclose(calibrated_effort, 0.9)

def normalize_log_weights(log_weights):
    """Normalize a finite-support log posterior."""
    log_weights = np.asarray(log_weights, dtype=float)
    finite_mask = np.isfinite(log_weights)

    if not np.any(finite_mask):
        return None

    maximum = float(np.max(log_weights[finite_mask]))

    shifted = np.zeros_like(log_weights)
    shifted[finite_mask] = np.exp(
        log_weights[finite_mask] - maximum
    )

    total = float(np.sum(shifted))

    if total <= 0.0 or not np.isfinite(total):
        return None

    return shifted / total


def candidate_outcome_log_likelihood(
    candidate_training_probabilities,
    outcomes,
):
    """
    Candidate-wise log likelihood of one complete binary outcome sequence.
    """
    outcomes = np.asarray(outcomes, dtype=int)
    candidate_training_probabilities = np.asarray(
        candidate_training_probabilities,
        dtype=float,
    )

    total_log_likelihood = np.zeros(
        candidate_training_probabilities.shape[1],
        dtype=float,
    )

    for round_index, outcome in enumerate(outcomes):
        probabilities = candidate_training_probabilities[
            round_index
        ]

        if outcome == 1:
            with np.errstate(divide="ignore"):
                contribution = np.log(probabilities)
        else:
            with np.errstate(divide="ignore", invalid="ignore"):
                contribution = np.log1p(-probabilities)

        total_log_likelihood += contribution

    return total_log_likelihood


def exact_effort_support_mask(
    candidate_effort_matrix,
    true_training_efforts,
    monitored_rounds,
):
    """
    Candidate points compatible with all exact monitored effort observations.
    """
    support = np.ones(
        candidate_effort_matrix.shape[1],
        dtype=bool,
    )

    support_sizes_by_round = []

    for round_number in monitored_rounds:
        workload_index = TRAINING_WORKLOAD_INDICES[
            round_number - 1
        ]

        observed_effort = float(
            true_training_efforts[round_number - 1]
        )

        support &= np.isclose(
            candidate_effort_matrix[workload_index],
            observed_effort,
            atol=1e-12,
            rtol=0.0,
        )

        support_sizes_by_round.append(int(np.sum(support)))

    return support, support_sizes_by_round


def sequence_probability(
    outcomes,
    true_training_probabilities,
):
    """Exact probability of one binary outcome sequence."""
    probability = 1.0

    for outcome, success_probability_value in zip(
        outcomes,
        true_training_probabilities,
    ):
        probability *= (
            success_probability_value
            if outcome == 1
            else 1.0 - success_probability_value
        )

    return float(probability)


def bernoulli_kl(
    true_probability,
    predicted_probability,
):
    """
    KL(Bernoulli(true) || Bernoulli(predicted)).
    """
    p = np.clip(
        np.asarray(true_probability, dtype=float),
        PROBABILITY_CLIP,
        1.0 - PROBABILITY_CLIP,
    )

    q = np.clip(
        np.asarray(predicted_probability, dtype=float),
        PROBABILITY_CLIP,
        1.0 - PROBABILITY_CLIP,
    )

    return (
        p * np.log(p / q)
        + (1.0 - p) * np.log((1.0 - p) / (1.0 - q))
    )


def posterior_entropy(posterior):
    positive = posterior > 0.0

    return float(
        -np.sum(
            posterior[positive]
            * np.log(posterior[positive])
        )
    )


def compute_predictive_metrics(
    posterior,
    predicted_curve,
    true_curve,
    capacity_vector,
    lambda_vector,
    true_capacity,
    true_lambda,
):
    """
    Predictive metrics for one posterior and one outcome sequence.
    """
    metrics = {
        "posterior_entropy": posterior_entropy(posterior),
        "normalized_posterior_entropy": (
            posterior_entropy(posterior)
            / math.log(len(posterior))
        ),
        "posterior_mean_capacity": float(
            np.dot(posterior, capacity_vector)
        ),
        "posterior_mean_lambda": float(
            np.dot(posterior, lambda_vector)
        ),
    }

    metrics["capacity_absolute_error"] = abs(
        metrics["posterior_mean_capacity"]
        - true_capacity
    )

    metrics["lambda_absolute_error"] = abs(
        metrics["posterior_mean_lambda"]
        - true_lambda
    )

    for evaluation_name, indices in EVALUATION_INDICES.items():
        true_values = true_curve[indices]
        predicted_values = predicted_curve[indices]

        absolute_error = np.abs(
            predicted_values - true_values
        )

        squared_error = (
            predicted_values - true_values
        ) ** 2

        kl_values = bernoulli_kl(
            true_values,
            predicted_values,
        )

        metrics[f"{evaluation_name}_mae"] = float(
            np.mean(absolute_error)
        )

        metrics[f"{evaluation_name}_rmse"] = float(
            np.sqrt(np.mean(squared_error))
        )

        metrics[f"{evaluation_name}_mean_kl"] = float(
            np.mean(kl_values)
        )

        metrics[
            f"{evaluation_name}_max_abs_error"
        ] = float(np.max(absolute_error))

    return metrics

class ProgressTracker:
    """
    Candidate-count-weighted progress indicator.
    """

    def __init__(
        self,
        total_units,
        report_every_percent=1.0,
    ):
        self.total_units = float(total_units)
        self.completed_units = 0.0
        self.report_every_percent = float(
            report_every_percent
        )
        self.next_report_percent = 0.0
        self.start_time = time.perf_counter()

    @staticmethod
    def format_seconds(seconds):
        if not np.isfinite(seconds):
            return "unknown"

        seconds = max(0, int(round(seconds)))
        hours, remainder = divmod(seconds, 3600)
        minutes, seconds = divmod(remainder, 60)

        if hours > 0:
            return (
                f"{hours:d}h "
                f"{minutes:02d}m "
                f"{seconds:02d}s"
            )

        if minutes > 0:
            return f"{minutes:d}m {seconds:02d}s"

        return f"{seconds:d}s"

    def update(self, units, context=""):
        self.completed_units += float(units)

        percent = min(
            100.0,
            100.0
            * self.completed_units
            / self.total_units,
        )

        if (
            percent + 1e-12
            < self.next_report_percent
            and percent < 100.0
        ):
            return

        elapsed = time.perf_counter() - self.start_time

        if self.completed_units > 0.0:
            estimated_total = (
                elapsed
                * self.total_units
                / self.completed_units
            )

            eta = max(0.0, estimated_total - elapsed)
        else:
            eta = math.inf

        suffix = f" | {context}" if context else ""

        print(
            f"Progress: {percent:6.2f}%"
            f" | elapsed {self.format_seconds(elapsed)}"
            f" | ETA {self.format_seconds(eta)}"
            f"{suffix}",
            flush=True,
        )

        while (
            self.next_report_percent
            <= percent + 1e-12
        ):
            self.next_report_percent += (
                self.report_every_percent
            )

)

convergence_summary_df[
    "absolute_aggregate_heldout_near_rmse_difference"
] = np.abs(
    convergence_summary_df[
        "heldout_near_rmse"
    ]
    - convergence_summary_df[
        "reference_aggregate_heldout_near_rmse"
    ]
)

convergence_summary_df[
    "absolute_aggregate_heldout_full_rmse_difference"
] = np.abs(
    convergence_summary_df[
        "heldout_full_rmse"
    ]
    - convergence_summary_df[
        "reference_aggregate_heldout_full_rmse"
    ]
)

convergence_summary_df[
    "zero_support_collapse"
] = np.isclose(
    convergence_summary_df[
        "support_collapse_rate"
    ],
    0.0,
)

convergence_summary_df[
    "reference_curve_tolerance_pass"
] = (
    convergence_summary_df[
        "reference_heldout_near_curve_rmse"
    ]
    <= REFERENCE_CURVE_RMSE_TOLERANCE
)

convergence_summary_df[
    "aggregate_rmse_tolerance_pass"
] = (
    convergence_summary_df[
        "absolute_aggregate_heldout_near_rmse_difference"
    ]
    <= AGGREGATE_RMSE_DIFFERENCE_TOLERANCE
)

convergence_summary_df[
    "basic_convergence_pass"
] = (
    convergence_summary_df[
        "zero_support_collapse"
    ]
    & convergence_summary_df[
        "reference_curve_tolerance_pass"
    ]
    & convergence_summary_df[
        "aggregate_rmse_tolerance_pass"
    ]
)


# =================================================================
# 17. PHASE-SHIFT SENSITIVITY
# =================================================================

phase_sensitivity_rows = []

for grid_size in [17, 33]:
    relevant_approximations = approximations_df[
        approximations_df["n_capacity"]
        == grid_size
    ]["approximation"].tolist()

    for method_name in METHOD_NAMES:
        subset = convergence_summary_df[
            (
                convergence_summary_df[
                    "approximation"
                ].isin(relevant_approximations)
            )
            & (
                convergence_summary_df[
                    "method"
                ] == method_name
            )
        ].copy()

        paired_subset = paired_gains_aggregate_df[
            (
                paired_gains_aggregate_df[
                    "approximation"
                ].isin(relevant_approximations)
            )
            & (
                paired_gains_aggregate_df[
                    "method"
                ] == method_name
            )
        ].copy()

        if subset.empty:
            continue

        row = {
            "grid_size_per_dimension": grid_size,
            "candidate_count": grid_size**2,
            "method": method_name,
            "phase_count": len(subset),
            "phases": "|".join(
                f"{value:.2f}"
                for value in sorted(
                    subset["phase"].tolist()
                )
            ),
            "heldout_near_rmse_min": float(
                subset[
                    "heldout_near_rmse"
                ].min()
            ),
            "heldout_near_rmse_max": float(
                subset[
                    "heldout_near_rmse"
                ].max()
            ),
            "heldout_near_rmse_phase_range": float(
                subset[
                    "heldout_near_rmse"
                ].max()
                - subset[
                    "heldout_near_rmse"
                ].min()
            ),
            "reference_curve_rmse_phase_range": float(
                subset[
                    "reference_heldout_near_curve_rmse"
                ].max()
                - subset[
                    "reference_heldout_near_curve_rmse"
                ].min()
            ),
            "support_collapse_rate_max": float(
                subset[
                    "support_collapse_rate"
                ].max()
            ),
        }

        if not paired_subset.empty:
            improvement_column = (
                "heldout_near_rmse_"
                "expected_improvement"
            )

            row[
                "paired_improvement_min"
            ] = float(
                paired_subset[
                    improvement_column
                ].min()
            )

            row[
                "paired_improvement_max"
            ] = float(
                paired_subset[
                    improvement_column
                ].max()
            )

            row[
                "paired_improvement_phase_range"
            ] = float(
                paired_subset[
                    improvement_column
                ].max()
                - paired_subset[
                    improvement_column
                ].min()
            )

        row["phase_tolerance_pass"] = bool(
            row[
                "heldout_near_rmse_phase_range"
            ]
            <= PHASE_RANGE_TOLERANCE
        )

        phase_sensitivity_rows.append(row)

phase_sensitivity_df = pd.DataFrame(
    phase_sensitivity_rows
)

    (
        int(row["candidate_count"]),

    for _, row in phase_sensitivity_df.iterrows()

] = midpoint_convergence_df.apply(

        True

        else phase_pass_lookup[
            (
                int(row["candidate_count"]),
                row["method"],
            )
        ]
    ),
    axis=1,
)

    "full_numerical_screen_pass"
] = (
    midpoint_convergence_df[
        "basic_convergence_pass"
    ]
    & midpoint_convergence_df[
        "phase_tolerance_pass"
    ]
